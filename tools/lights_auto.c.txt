/* the automatic lights: burning torches on the walls of rooms. Candidates: open tiles beside a wall in a room
 * (at least 13 of the 5 x 5 round them open: corridors stay dark, for the player's torch), the larger rooms
 * first; each kept only when no torch is within 5 tiles. The torch hangs on the wall, about 0.8 of a tile
 * above the floor, facing into the room. */
#define AUTO_LIGHTS_HEAD "# uo-new wall torches v2"
static bool write_auto(const char *path, const Level *lv) {
    typedef struct {
        int x, z, e, score;
    } Cand;
    static Cand c[4096 * 4];
    int nc = 0;
    static const int DX[4] = {0, 1, 0, -1}, DZ[4] = {-1, 0, 1, 0};
    for (int z = 0; z < 64; z++)
        for (int x = 0; x < 64; x++) {
            const Tile *t = &lv->tile[z * 64 + x];
            if (t->type != T_OPEN) continue;
            int open = 0;
            for (int dz = -2; dz <= 2; dz++)
                for (int dx = -2; dx <= 2; dx++) {
                    int xx = x + dx, zz = z + dz;
                    if (xx >= 0 && xx < 64 && zz >= 0 && zz < 64 && lv->tile[zz * 64 + xx].type != T_SOLID) open++;
                }
            if (open < 13) continue;
            for (int e = 0; e < 4; e++) {
                int xx = x + DX[e], zz = z + DZ[e];
                bool wall = xx < 0 || xx > 63 || zz < 0 || zz > 63 || lv->tile[zz * 64 + xx].type == T_SOLID;
                /* a wall the whole way up from this floor (not a step) */
                if (!wall) continue;
                c[nc++] = (Cand){x, z, e, open * 4096 + (int)(hash((u32)((z * 64 + x) * 4 + e) * 2654435761u) & 4095)};
            }
        }
    for (int i = 1; i < nc; i++) /* the larger rooms first */
        for (int j = i; j > 0 && c[j].score > c[j - 1].score; j--) {
            Cand t = c[j];
            c[j] = c[j - 1], c[j - 1] = t;
        }
    static float px[MAX_LIGHTS], pz[MAX_LIGHTS];
    int np = 0;
    char *text = arena_alloc(1 << 17);
    size_t tn = (size_t)snprintf(text, 1 << 17,
                                 AUTO_LIGHTS_HEAD " for level %d: burning torches on the walls of rooms, placed by uo-new (the larger\r\n"
                                 "# rooms first, about 5 tiles apart; corridors stay dark). Edit, move, recolour or delete lines freely;\r\n"
                                 "# delete the file to have them placed again.\r\n"
                                 "#   walltorch LEVEL X Z HEIGHT  R G B  INTENSITY RANGE  FX FZ   (FX FZ: the way it faces, out of the wall)\r\n",
                                 lv->number);
    for (int i = 0; i < nc && np < MAX_LIGHTS - 64; i++) {
        /* on the wall, a little out from it */
        float fx = (float)-DX[c[i].e], fz = (float)-DZ[c[i].e];
        float x = (float)c[i].x + 0.5f - fx * 0.40f, z = (float)c[i].z + 0.5f - fz * 0.40f;
        bool near = false;
        for (int k = 0; k < np && !near; k++) near = (px[k] - x) * (px[k] - x) + (pz[k] - z) * (pz[k] - z) < 25; /* 5 tiles */
        if (near) continue;
        px[np] = x, pz[np] = z, np++;
        const Tile *t = &lv->tile[c[i].z * 64 + c[i].x];
        float h = MIN((float)t->height + 3.2f, (float)CEIL_STEPS - 0.6f);
        u32 r = hash((u32)(c[i].z * 64 + c[i].x) + 77u);
        float g = 0.66f + (float)(r & 255) / 255.0f * 0.10f, b = 0.36f + (float)((r >> 8) & 255) / 255.0f * 0.12f;
        tn += (size_t)snprintf(text + tn, (1 << 17) - tn, "walltorch %d %.2f %.2f %.1f  1.0 %.2f %.2f  1.3 6.0  %.0f %.0f\r\n", lv->number, x, z, h,
                               g, b, fx, fz);
    }
    make_dirs_for(path);
    return file_write(path, text, tn);
}

/* the automatic file is (re)made when missing, or when it is the engine's own from before the wall torches */
static bool auto_current(const char *path) {
    FILE *f = fopen(path, "rb");
    if (!f) return false;
    char line[96] = {0};
    size_t n = fread(line, 1, sizeof line - 1, f);
    fclose(f);
    line[n] = 0;
    return strncmp(line, "# Lights placed automatically for level", 39) != 0;
}
